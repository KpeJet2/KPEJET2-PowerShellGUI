2) Expected Pipeline Counts

function getPipelineExpectedForType(type) {
  if (!pipeData) return null;
  if (type === 'bug') return (pipeData.bugs||[]).length + (pipeData.bugs2FIX||[]).length;
  if (type === 'feature') return (pipeData.featureRequests||[]).length;
  if (type === 'todo') return (pipeData.todos||[]).length + (pipeData.items2ADD||[]).length;
  return null;
}

5) Delta Parser Helper

function metricDeltaFromNote(noteId) {
  var txt = String(document.getElementById(noteId).textContent || '');
  var m = txt.match(/delta\s*=\s*([+-]?\d+)/i);
  return m ? Number(m[1]) : null;
}
